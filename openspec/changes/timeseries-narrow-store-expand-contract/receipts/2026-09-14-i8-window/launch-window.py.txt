def digest(path):

def role_url(name, user):
    path = ACTIVE / "infra/env" / name
    assert not path.is_symlink() and path.stat().st_uid == 1005 and path.stat().st_mode & 0o777 == 0o600
    values = []
    for line in path.read_text().splitlines():
        if line.strip().startswith("DATABASE_URL="):
            values.extend(shlex.split(line.strip().split("=", 1)[1], comments=True))
    assert len(values) == 1
    value = values[0]
    parsed = urllib.parse.urlsplit(value)
    assert (parsed.username, parsed.hostname, parsed.port, parsed.path) == (user, "127.0.0.1", 55432, "/nhms")
    return value
