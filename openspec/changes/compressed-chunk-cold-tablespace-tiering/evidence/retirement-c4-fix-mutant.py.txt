Parent runs this on isolated node27. Never mutate test expectations. Each
selected guard-family removal must kill the corresponding new row; sha-only

checks must kill the matching F3 row; closed-stdout reds against original CLI.
"""

PYTEST = [sys.executable, "-m", "pytest", "-q"]


def _run(tests: list[str]) -> subprocess.CompletedProcess[str]:
    return subprocess.run([*PYTEST, *tests], cwd=REPO, text=True, timeout=180, capture_output=True)


def _assert_red(result: subprocess.CompletedProcess[str], label: str) -> None:
    if result.returncode != 1 or "FAILED " not in result.stdout or "ERROR collecting" in result.stdout or "ERROR at setup" in result.stdout:
        raise SystemExit(f"{label} did not produce assertion-red:\n{result.stdout}\n{result.stderr}")
    print(result.stdout)


def _replace_once(path: Path, old: str, new: str) -> None:
    text = path.read_text()
    if old not in text:
        raise SystemExit(f"anchor missing in {path}: {old!r}")
    if text.count(old) != 1:
        raise SystemExit(f"anchor not unique in {path}: {old!r} count={text.count(old)}")
    path.write_text(text.replace(old, new, 1))


def mutate(path: Path, old: str, new: str, tests: list[str], label: str) -> None:
    original = path.read_bytes()
    try:
        _replace_once(path, old, new)
        red = _run(tests)
        _assert_red(red, label)
    finally:
        path.write_bytes(original)
        if path.read_bytes() != original:
            raise SystemExit(f"{path} was not restored exactly")
    print(f"RED {label}")
