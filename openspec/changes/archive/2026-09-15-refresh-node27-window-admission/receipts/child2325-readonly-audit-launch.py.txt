"""Qualification only: run frozen415 audit with unchanged private env; never invoke a unit."""

import hashlib
import json
import os
import stat
import subprocess
from pathlib import Path

from psycopg2.extensions import parse_dsn

TARGET = Path("/home/nwm/NWM-i8-target-415cbd1e")
SHA = "415cbd1e9d0eee39ba0dfb623a586b02cbb340f2"
ENV_FILE = Path("/home/nwm/NWM/infra/env/node27-resource-governance.env")
ROOT = Path("/home/nwm/tmp/issue2325-audit-b55c7ef9b43e")
PGDATA = Path("/data/GHDC/nhms-primary/pgdata")


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    os.umask(0o077)
    assert os.getuid() == 1005
    assert subprocess.check_output(["git", "-C", str(TARGET), "rev-parse", "HEAD"], text=True).strip() == SHA
    assert not subprocess.check_output(["git", "-C", str(TARGET), "status", "--porcelain"], text=True).strip()
    sources = {}
    for relative in (
        "scripts/node27_resource_governance.py",
        "packages/common/node27_resource_governance_collection.py",
    ):
        path = TARGET / relative
        expected = subprocess.check_output(["git", "-C", str(TARGET), "show", SHA + ":" + relative])
        assert path.read_bytes() == expected
        sources[relative] = digest(path)
    info = ENV_FILE.lstat()
    assert stat.S_ISREG(info.st_mode) and info.st_uid == 1005 and stat.S_IMODE(info.st_mode) == 0o600
    env_hash = digest(ENV_FILE)
    info = PGDATA.stat()
    pgdata_identity = [info.st_dev, info.st_ino, info.st_uid, info.st_gid, stat.S_IMODE(info.st_mode)]
    assert pgdata_identity == [2304, 126746626, 1005, 1005, 0o700]
    primary = json.loads(subprocess.check_output(["docker", "inspect", "nhms-db"]))[0]
    assert primary["Id"] == "5cfa71472de87f926d1fd7c085edc3ff6ab8cea445e899be3b2676bb2d607e5f"
    assert (
        primary["State"]["Running"]
        and primary["Image"] == "sha256:ad39c4fbc5c44557db1e16af10ec11e3ab12d0a472374f39aaba06ad9ca2640e"
    )
    container_pgdata = Path("/home/postgres/pgdata/data")
    assert any(
        container_pgdata.is_relative_to(mount["Destination"])
        and Path(mount["Source"]) / container_pgdata.relative_to(mount["Destination"]) == PGDATA
        for mount in primary["Mounts"]
    )
    ROOT.mkdir(mode=0o700)
    loaded = subprocess.run(
        ["/bin/bash", "--noprofile", "--norc", "-c", 'set -a; . "$1"; env -0', "audit-env", str(ENV_FILE)],
        capture_output=True,
        check=True,
    )
    environment = dict(item.split("=", 1) for item in loaded.stdout.decode().split("\0") if "=" in item)
    assert digest(ENV_FILE) == env_hash
    dsn = parse_dsn(environment["DATABASE_URL"])
    assert dsn.get("host") in ("127.0.0.1", "localhost")
    assert dsn.get("port", "5432") == "55432" and dsn.get("dbname") == "nhms"
    assert dsn.get("user") == "nhms_display_ro"
    environment.pop("PYTHONPATH", None)
    environment.pop("PYTHONHOME", None)
    environment["TMPDIR"] = "/home/nwm/tmp"
    summary = ROOT / "summary.json"
    command = [
        str(TARGET / ".venv/bin/python"),
        "-B",
        str(TARGET / "scripts/node27_resource_governance.py"),
        "--summary-path",
        str(summary),
        "--quiet",
    ]
    with (ROOT / "stdout").open("xb") as out, (ROOT / "stderr").open("xb") as err:
        result = subprocess.run(command, cwd=TARGET, env=environment, stdout=out, stderr=err, timeout=240)
    receipt = json.loads(summary.read_text())
    assert digest(ENV_FILE) == env_hash
    assert receipt["execution_mode"] == "read_only_audit"
    assert receipt["paths"]["pgdata_root"] == str(PGDATA)
    assert receipt["safety"]["destructive_actions_enabled"] is False
    critical = [row.get("code") for row in receipt.get("recommendations", []) if row.get("severity") == "critical"]
    proof = dict(
        deployment_sha=SHA,
        sources=sources,
        command=command,
        exit=result.returncode,
        summary_path=str(summary),
        summary_sha256=digest(summary),
        environment_path=str(ENV_FILE),
        environment_sha256=env_hash,
        environment_unchanged=True,
        pgdata_identity=pgdata_identity,
        container_id=primary["Id"],
        container_image=primary["Image"],
        status=receipt.get("status"),
        critical_codes=critical,
        logs={str(ROOT / name): digest(ROOT / name) for name in ("stdout", "stderr")},
        service_start_requested=False,
        production_mutation=False,
    )
    with (ROOT / "qualification.json").open("x") as stream:
        json.dump(proof, stream, indent=2)
    print(json.dumps(proof), flush=True)
    return result.returncode


if __name__ == "__main__":
    raise SystemExit(main())
